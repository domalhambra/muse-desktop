# mMac M0 (the spike) Implementation Plan

**Status:** built on 2026-10-21 on the `mac-app` branch. The code differs from this plan where review asked for it. For example, the wire test has 11 tests, the parity test has 6 and `swift test` runs 22. The decisions behind each difference are in `docs/decisions.md`, under "2026-10-21: mMac M0". The work M0 leaves open is in "Carried into M1" at the end.

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:subagent-driven-development (recommended) or superpowers-extended-cc:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A headless Node front door and a bare SwiftUI window that connect to Muse, list conversations and stream one echo turn, proved by `scripts/verify-headless.ts`.

**Architecture:** `src/headless/server.ts` runs the fork's existing core in one plain Node process and speaks one JSON object per line over stdin and stdout (Request, Reply, Event, Ask). A Swift package, `MuseBackend`, starts that process and turns the lines into typed Swift calls and events. The `mMac` app is a thin SwiftUI window over `MuseBackend`. Spec: `docs/superpowers/specs/2026-10-21-mac-app-design.md`.

**Tech Stack:** TypeScript, `tsx`, `zod`, `@muse-code/sdk` (through `src/main/muse-host.ts`), Vitest; Swift 6, SwiftUI, Swift Testing, XcodeGen, Xcode 27.

### Task 1: The wire module

**Goal:** `src/headless/wire.ts` parses and encodes the four message kinds and dispatches Requests, with unit tests.

**Files:**

- Create: `src/headless/wire.ts`
- Test: `tests/unit/headless-wire.test.ts`

**Acceptance Criteria:**

- [ ] A Request for a known method gets `{"id":…,"ok":true,"value":…}`. A method that returns nothing gets `"value":null`.
- [ ] A throwing or unknown method gets `{"id":…,"ok":false,"error":"…"}` with the message `messageOf()` gives.
- [ ] A line that is not valid JSON, or not a valid Request, produces one `{"event":{"type":"error",…}}` line and no exception.
- [ ] `ask()` writes `{"ask":n,…}` and resolves with the `choice` of the matching `{"answer":n,…}`. `cancelAsks()` resolves every open Ask with `-1`.
- [ ] `drain()` resolves only after every request received so far has written its Reply, so quitting never drops a Reply.
- [ ] Every written line is one JSON object followed by exactly one `\n`.

```ts
import { describe, expect, it } from 'vitest'
import { Wire, type Method } from '../../src/headless/wire'

function harness(methods: Record<string, Method> = {}) {
  const lines: string[] = []
  const wire = new Wire((line) => lines.push(line), new Map(Object.entries(methods)))
  const sent = () => lines.map((line) => JSON.parse(line) as Record<string, unknown>)
  return { wire, lines, sent }
}

describe('headless wire', () => {
  it('replies with the value a method returns', async () => {
    const { wire, sent } = harness({ add: (a, b) => (a as number) + (b as number) })
    await wire.receive(JSON.stringify({ id: 7, method: 'add', args: [2, 3] }))
    expect(sent()).toEqual([{ id: 7, ok: true, value: 5 }])
  })

  it('replies null when a method returns nothing', async () => {
    const { wire, sent } = harness({ quiet: () => undefined })
    await wire.receive(JSON.stringify({ id: 1, method: 'quiet' }))
    expect(sent()).toEqual([{ id: 1, ok: true, value: null }])
  })

  it('replies with the error message when a method throws or does not exist', async () => {
    const { wire, sent } = harness({
      fail: () => {
        throw new Error('Send start first.')
      },
    })
    await wire.receive(JSON.stringify({ id: 1, method: 'fail', args: [] }))
    await wire.receive(JSON.stringify({ id: 2, method: 'missing', args: [] }))
    expect(sent()).toEqual([
      { id: 1, ok: false, error: 'Send start first.' },
      { id: 2, ok: false, error: 'Unknown method: missing' },
    ])
  })

  it('reports an unreadable line as an error event and ignores blank lines', async () => {
    const { wire, sent } = harness()
    await wire.receive('not json')
    await wire.receive(JSON.stringify({ id: 'x', method: 'sessions' }))
    await wire.receive('   ')
    const lines = sent()
    expect(lines).toHaveLength(2)
    for (const line of lines) expect(line.event).toMatchObject({ type: 'error' })
  })

  it('sends an Ask and resolves it with the answer', async () => {
    const { wire, sent } = harness()
    const choice = wire.ask({
      kind: 'confirm',
      title: 'Trust this project?',
      message: 'Work with Muse in “p”?',
      buttons: ['Trust project', 'Cancel'],
    })
    expect(sent()).toEqual([
      {
        ask: 1,
        kind: 'confirm',
        title: 'Trust this project?',
        message: 'Work with Muse in “p”?',
        buttons: ['Trust project', 'Cancel'],
      },
    ])
    await wire.receive(JSON.stringify({ answer: 1, choice: 0 }))
    await expect(choice).resolves.toBe(0)
  })

  it('answers every open Ask with -1 on quit', async () => {
    const { wire } = harness()
    const first = wire.ask({ kind: 'confirm', title: 'A', message: 'A', buttons: ['OK', 'Cancel'] })
    const second = wire.ask({
      kind: 'confirm',
      title: 'B',
      message: 'B',
      buttons: ['OK', 'Cancel'],
    })
    wire.cancelAsks()
    await expect(first).resolves.toBe(-1)
    await expect(second).resolves.toBe(-1)
  })

  it('drains requests still in flight', async () => {
    let finish = () => {}
    const { wire, sent } = harness({
      slow: () =>
        new Promise<void>((resolve) => {
          finish = () => resolve()
        }),
    })
    void wire.receive(JSON.stringify({ id: 1, method: 'slow' }))
    const drained = wire.drain()
    expect(sent()).toEqual([])
    finish()
    await drained
    expect(sent()).toEqual([{ id: 1, ok: true, value: null }])
  })

  it('writes one JSON object per line', () => {
    const { wire, lines } = harness()
    wire.emit({ type: 'sessions-changed' })
    expect(lines).toEqual(['{"event":{"type":"sessions-changed"}}\n'])
  })
})
```

```ts
import { z } from 'zod'
import type { AppEvent } from '../shared/protocol'
import { messageOf } from '../main/muse-host'

// mMac and this backend speak one JSON object per line over stdin and stdout.
// Request (app → backend), Reply and Event (backend → app), and Ask (backend →
// app, then answered). See docs/superpowers/specs/2026-10-21-mac-app-design.md.

/** Change this, and `backendProtocol` in mMac's `Messages.swift`, when a message shape changes. */
export const HEADLESS_PROTOCOL = 1

const requestSchema = z.object({
  id: z.number().int().nonnegative(),
  method: z.string().min(1).max(100),
  args: z.array(z.unknown()).max(10).default([]),
})
const askAnswerSchema = z.object({
  answer: z.number().int().positive(),
  choice: z.number().int().min(-1),
})

export type Method = (...args: unknown[]) => unknown

/** A native alert in mMac. The last button cancels. */
export type Ask = {
  kind: 'confirm'
  title: string
  message: string
  detail?: string
  buttons: string[]
}

type Outgoing =
  | { id: number; ok: true; value: unknown }
  | { id: number; ok: false; error: string }
  | { event: AppEvent }
  | ({ ask: number } & Ask)

function encode(message: Outgoing) {
  return `${JSON.stringify(message)}\n`
}

export class Wire {
  private asks = new Map<number, (choice: number) => void>()
  private nextAsk = 1
  private inFlight = new Set<Promise<void>>()

  constructor(
    private write: (line: string) => void,
    private methods: Map<string, Method>,
  ) {}

  emit(event: AppEvent) {
    this.write(encode({ event }))
  }

  /** Shows a native alert in mMac and resolves with the chosen button's index. */
  ask(ask: Ask): Promise<number> {
    const id = this.nextAsk++
    return new Promise((resolve) => {
      this.asks.set(id, resolve)
      this.write(encode({ ask: id, ...ask }))
    })
  }

  /** Answers every open Ask with -1, the cancel choice. Used on quit. */
  cancelAsks() {
    for (const [id, resolve] of this.asks) {
      this.asks.delete(id)
      resolve(-1)
    }
  }

  /** Handles one line from mMac. Never rejects. */
  receive(line: string): Promise<void> {
    const work = this.handle(line)
    this.inFlight.add(work)
    void work.finally(() => this.inFlight.delete(work))
    return work
  }

  /** Resolves when every request received so far has written its Reply. */
  async drain() {
    await Promise.allSettled([...this.inFlight])
  }

  private async handle(line: string) {
    if (!line.trim()) return
    let value: unknown
    try {
      value = JSON.parse(line)
    } catch {
      this.emit({ type: 'error', message: 'mMac sent a line that is not JSON.' })
      return
    }
    if (value !== null && typeof value === 'object' && 'answer' in value) {
      const answer = askAnswerSchema.safeParse(value)
      if (!answer.success) {
        this.emit({ type: 'error', message: 'mMac sent an Ask answer the backend could not read.' })
        return
      }
      const resolve = this.asks.get(answer.data.answer)
      this.asks.delete(answer.data.answer)
      resolve?.(answer.data.choice)
      return
    }
    const request = requestSchema.safeParse(value)
    if (!request.success) {
      this.emit({ type: 'error', message: 'mMac sent a request the backend could not read.' })
      return
    }
    const { id, method, args } = request.data
    try {
      const action = this.methods.get(method)
      if (!action) throw new Error(`Unknown method: ${method}`)
      this.write(encode({ id, ok: true, value: (await action(...args)) ?? null }))
    } catch (error) {
      this.write(encode({ id, ok: false, error: messageOf(error) }))
    }
  }
}
```

### Task 2: The headless front door

**Goal:** `src/headless/server.ts` serves the M0 methods (`hello`, `start`, `connect`, `add-project`, `sessions`, `create`, `open`, `send`, `cancel`, `shutdown`) over the wire and quits cleanly.

**Files:**

- Create: `src/headless/server.ts`
- Modify: `package.json` (the `scripts` block)

**Acceptance Criteria:**

- [ ] stdout carries only protocol lines. `console.log`, `info`, `debug` and `warn` go to stderr.
- [ ] Any method before `hello` fails with "Send hello first." Any method except `hello`, `start` and `shutdown` before `start` fails with "Send start first."
- [ ] `hello` with another protocol number replies with an error and the process exits with code 2.
- [ ] `create` on an untrusted project sends the trust Ask. Choice 0 trusts it in mMac's preferences. Any other choice fails with "Project trust was cancelled. No session was started."
- [ ] `shutdown`, stdin closing and SIGTERM each close `muse serve`, wait for preference writes and exit with code 0.
- [ ] `npm run headless` starts the backend.

**Verify:** `npm run typecheck` → no errors, then the smoke check in Step 3 → one reply line and exit code 0.

**Steps:**

- [ ] **Step 1: Write the server**

Create `src/headless/server.ts`:

```ts
import { mkdir, readFile, realpath, stat } from 'node:fs/promises'
import { basename, join } from 'node:path'
import { createInterface } from 'node:readline'
import { z } from 'zod'
import {
  effortSchema,
  preferencesSchema,
  type AppEvent,
  type Preferences,
} from '../shared/protocol'
import { MuseHost, messageOf } from '../main/muse-host'
import { readPreferences, writePreferences } from '../main/preferences'
import { publish } from '../main/event-bus'
import { projectRoot, worktreeParent } from '../shared/worktree'
import { HEADLESS_PROTOCOL, Wire, type Method } from './wire'

// mMac's backend: the fork's core in one plain Node process, driven over stdin
// and stdout. It repeats the wiring of src/main/index.ts instead of sharing it,
// so upstream syncs of index.ts stay clean. tests/unit/headless-parity.test.ts
// catches drift between the two.

// stdout carries only protocol lines. Anything a library logs goes to stderr.
const writeOut = process.stdout.write.bind(process.stdout)
for (const name of ['log', 'info', 'debug', 'warn'] as const) console[name] = console.error

const { version } = JSON.parse(
  await readFile(new URL('../../package.json', import.meta.url), 'utf8'),
) as { version: string }
const id = z.string().min(1).max(200)
const pathSchema = z.string().min(1).max(4096)
const methods = new Map<string, Method>()
const wire = new Wire((line) => {
  writeOut(line)
}, methods)
let greeted = false
let preferences: Preferences | undefined
let preferencePath = ''
let preferenceWrites: Promise<void> = Promise.resolve()
let quitting = false

function emit(event: AppEvent) {
  publish(event)
  wire.emit(event)
}
const muse = new MuseHost(emit)

function method(name: string, action: Method) {
  methods.set(name, async (...args: unknown[]) => {
    if (name !== 'hello' && !greeted) throw new Error('Send hello first.')
    if (!['hello', 'start', 'shutdown'].includes(name) && !preferences)
      throw new Error('Send start first.')
    return action(...args)
  })
}

function current() {
  if (!preferences) throw new Error('Send start first.')
  return preferences
}

async function savePreferences(patch: Partial<Preferences>) {
  const write = preferenceWrites.then(async () => {
    const snapshot = preferencesSchema.parse({ ...current(), ...patch })
    await writePreferences(preferencePath, snapshot)
    preferences = snapshot
    emit({ type: 'preferences', preferences: snapshot })
    return snapshot
  })
  preferenceWrites = write.then(
    () => {},
    () => {},
  ) // A failed write must not block later saves.
  return write
}

async function knownProject(value: unknown) {
  const project = await realpath(pathSchema.parse(value))
  const resolvedRecent = await Promise.all(
    current().recentProjects.map(async (entry) => {
      try {
        return await realpath(entry)
      } catch {
        return entry
      }
    }),
  )
  // A Muse-created worktree (`<project>/.muse/worktrees/<name>`) belongs to its project.
  const candidates = [project, worktreeParent(project)].filter((entry) => entry !== null)
  if (!resolvedRecent.some((entry) => candidates.includes(entry)))
    throw new Error('Choose this project using Open project first.')
  return project
}

async function trustProject(project: string) {
  const { trustedProjects } = current()
  if (trustedProjects.includes(project)) return
  const parent = worktreeParent(project)
  if (parent && trustedProjects.includes(parent)) return
  const choice = await wire.ask({
    kind: 'confirm',
    title: 'Trust this project?',
    message: `Work with Muse in “${basename(project)}”?`,
    detail: `Muse can read and edit this project and use its skills, instructions, and hooks. Its existing sandbox and tool approval settings remain active.\n\n${project}\n\nOnly continue with a project you trust.`,
    buttons: ['Trust project', 'Cancel'],
  })
  if (choice !== 0) throw new Error('Project trust was cancelled. No session was started.')
  await savePreferences({ trustedProjects: [...current().trustedProjects, project] })
}

async function sessionProject(sessionId: string) {
  const session = muse.getSession(sessionId)
  if (!session.workspaceRoot) throw new Error('This conversation has no project folder.')
  const project = await realpath(session.workspaceRoot)
  await trustProject(project)
  return project
}

/** Waits for every queued stdout write, then exits. */
function exit(code: number) {
  process.stdout.write('', () => process.exit(code))
}

async function quit(code: number) {
  if (quitting) return
  quitting = true
  wire.cancelAsks()
  try {
    // Closing Muse makes waiting requests fail, so drain() cannot wait forever.
    await Promise.all([muse.close(), preferenceWrites, wire.drain()])
  } catch (error) {
    console.error(`The mMac backend did not shut down cleanly: ${messageOf(error)}`)
    code ||= 1
  }
  exit(code)
}

method('hello', (value) => {
  const theirs = z.number().int().parse(value)
  if (theirs !== HEADLESS_PROTOCOL) {
    setImmediate(() => void quit(2))
    throw new Error(
      `mMac speaks backend protocol ${theirs}, but this backend speaks ${HEADLESS_PROTOCOL}. Build mMac and the backend from the same checkout.`,
    )
  }
  greeted = true
  return { protocol: HEADLESS_PROTOCOL, version }
})

method('start', async (value) => {
  if (preferences) throw new Error('The backend has already started.')
  const { dataFolder } = z.object({ dataFolder: pathSchema }).parse(value)
  await mkdir(dataFolder, { recursive: true, mode: 0o700 })
  preferencePath = join(dataFolder, 'preferences.json')
  preferences = await readPreferences(preferencePath)
  const status = await muse.connect(preferences.cliPath)
  return { preferences, status }
})

method('connect', async () => {
  if (muse.hasActiveTurns()) {
    const choice = await wire.ask({
      kind: 'confirm',
      title: 'Reconnect while Muse is working?',
      message:
        'Reconnecting closes the current CLI host and interrupts its active turns. Saved conversations can be resumed afterward.',
      buttons: ['Reconnect', 'Cancel'],
    })
    if (choice !== 0) return muse.status
  }
  return muse.connect(current().cliPath)
})

method('add-project', async (value) => {
  const project = await realpath(pathSchema.parse(value))
  if (!(await stat(project)).isDirectory()) throw new Error('Choose a folder, not a file.')
  await savePreferences({
    recentProjects: [
      project,
      ...current().recentProjects.filter((entry) => entry !== project),
    ].slice(0, 30),
  })
  return project
})

method('sessions', async (cursor) => {
  const result = await muse.list(z.string().optional().parse(cursor))
  const roots = [
    ...new Set(
      result.sessions
        .filter((session) => session.kind !== 'subagent')
        .map((session) => session.workspaceRoot)
        .filter((root): root is string => Boolean(root)),
    ),
  ]
  const discovered = await Promise.all(
    roots.map(async (root) => {
      try {
        const canonical = await realpath(root)
        return (await stat(canonical)).isDirectory() ? canonical : null
      } catch {
        return null
      }
    }),
  )
  const recentProjects = [
    ...new Set([
      ...current().recentProjects,
      ...discovered.filter((root): root is string => root !== null),
    ]),
  ].slice(0, 30)
  if (recentProjects.length !== current().recentProjects.length)
    await savePreferences({ recentProjects })
  return result
})

method('create', async (project, modelId, options) => {
  const root = await knownProject(project)
  await trustProject(root)
  const { worktree } =
    z.object({ worktree: z.boolean().optional() }).optional().parse(options) ?? {}
  return muse.create(root, z.string().optional().parse(modelId), { worktree })
})

method('open', async (value) => {
  const view = await muse.open(id.parse(value))
  // Record a worktree session's parent project, not each worktree folder.
  const root = view.session.workspaceRoot && projectRoot(view.session.workspaceRoot)
  if (root && !current().recentProjects.includes(root))
    await savePreferences({ recentProjects: [root, ...current().recentProjects].slice(0, 30) })
  return view
})

method('send', async (value) => {
  // M0 sends text only. Attachments come with the attachment methods in M4.
  const input = z
    .object({
      sessionId: id,
      text: z.string().max(200000),
      attachments: z.array(id).max(0),
      effort: effortSchema,
      mode: z.enum(['queue', 'steer']),
    })
    .parse(value)
  if (!input.text.trim()) throw new Error('Write a message first.')
  await sessionProject(input.sessionId)
  await muse.send(
    input.sessionId,
    [{ type: 'text', text: input.text }],
    input.effort,
    input.mode,
    input.text,
  )
})

method('cancel', (value) => muse.cancel(id.parse(value)))

method('shutdown', async () => {
  if (muse.hasActiveTurns()) {
    const choice = await wire.ask({
      kind: 'confirm',
      title: 'Quit while Muse is working?',
      message: 'Quitting interrupts active turns.',
      buttons: ['Quit', 'Keep working'],
    })
    if (choice !== 0) return false
  }
  setImmediate(() => void quit(0))
  return true
})

const lines = createInterface({ input: process.stdin, crlfDelay: Infinity })
lines.on('line', (line) => {
  if (!quitting) void wire.receive(line)
})
// mMac closed stdin, quit or crashed.
lines.on('close', () => void quit(0))
process.on('SIGTERM', () => void quit(0))
```

```swift
import Foundation
@testable import MuseBackend

/// A fake backend. Each line mMac sends goes to `script`, which returns the lines to answer with.
final class ScriptedBackend: Transport, @unchecked Sendable {
  let lines: AsyncStream<Data>
  private let continuation: AsyncStream<Data>.Continuation
  private let script: @Sendable ([String: JSONValue]) -> [String]
  private let lock = NSLock()
  private var received: [[String: JSONValue]] = []
  private var stopped = false

  init(script: @escaping @Sendable ([String: JSONValue]) -> [String] = { _ in [] }) {
    let stream = AsyncStream.makeStream(of: Data.self)
    lines = stream.stream
    continuation = stream.continuation
    self.script = script
  }

  /// Every message mMac sent, decoded.
  var messages: [[String: JSONValue]] { lock.withLock { received } }
  var forceStopped: Bool { lock.withLock { stopped } }

  func send(_ line: Data) throws {
    let message = try JSONDecoder().decode([String: JSONValue].self, from: line)
    lock.withLock { received.append(message) }
    for reply in script(message) { push(reply) }
  }

  /// Sends one line to mMac, as the backend would.
  func push(_ line: String) { continuation.yield(Data(line.utf8)) }

  /// Ends the stream, as the backend does when it exits.
  func close() { continuation.finish() }

  func forceStop() {
    lock.withLock { stopped = true }
    continuation.finish()
  }
}
```

```swift
import Foundation

/// Any JSON value. Request arguments and untyped replies travel as this.
public enum JSONValue: Codable, Equatable, Sendable {
  case null
  case bool(Bool)
  case number(Double)
  case string(String)
  case array([JSONValue])
  case object([String: JSONValue])

  public init(from decoder: Decoder) throws {
    let container = try decoder.singleValueContainer()
    if container.decodeNil() {
      self = .null
    } else if let value = try? container.decode(Bool.self) {
      self = .bool(value)
    } else if let value = try? container.decode(Double.self) {
      self = .number(value)
    } else if let value = try? container.decode(String.self) {
      self = .string(value)
    } else if let value = try? container.decode([JSONValue].self) {
      self = .array(value)
    } else {
      self = .object(try container.decode([String: JSONValue].self))
    }
  }

  public func encode(to encoder: Encoder) throws {
    var container = encoder.singleValueContainer()
    switch self {
    case .null: try container.encodeNil()
    case .bool(let value): try container.encode(value)
    case .number(let value): try container.encode(value)
    case .string(let value): try container.encode(value)
    case .array(let value): try container.encode(value)
    case .object(let value): try container.encode(value)
    }
  }

  /// Decodes this value as `type`, the way a JSON reply would decode.
  func decode<T: Decodable>(_ type: T.Type) throws -> T {
    try JSONDecoder().decode(type, from: JSONEncoder().encode(self))
  }
```

```swift
  public let ask: Int
  public let kind: String
  public let title: String
  public let message: String
  public let detail: String?
  public let buttons: [String]

struct OutgoingRequest: Encodable {
  let id: Int
  let method: String
  let args: [JSONValue]
}

struct OutgoingAnswer: Encodable {
  let answer: Int
  let choice: Int
}

enum Incoming {
  case reply(id: Int, result: Result<JSONValue, BackendError>)
  case event(AppEvent)
  case ask(Ask)

  private struct EventLine: Decodable { let event: AppEvent }

  static func decode(_ line: Data) throws -> Incoming {
    let decoder = JSONDecoder()
    let object = try decoder.decode([String: JSONValue].self, from: line)
    if case .number(let id)? = object["id"] {
      if object["ok"] == .bool(true) { return .reply(id: Int(id), result: .success(object["value"] ?? .null)) }
      if case .string(let message)? = object["error"] { return .reply(id: Int(id), result: .failure(.backend(message))) }
      return .reply(id: Int(id), result: .failure(.unreadable("A reply had no result.")))
    }
    if object["event"] != nil { return .event(try decoder.decode(EventLine.self, from: line).event) }
    if object["ask"] != nil { return .ask(try decoder.decode(Ask.self, from: line)) }
    throw BackendError.unreadable("A line was of no known kind.")
  }
}

/// Splits a byte stream into lines at each newline. The newline is not kept.
struct LineSplitter {
  private var buffer = Data()

  mutating func append(_ data: Data) -> [Data] {
    buffer.append(data)
    var lines: [Data] = []
    while let index = buffer.firstIndex(of: 0x0A) {
      lines.append(Data(buffer[buffer.startIndex..<index]))
      buffer.removeSubrange(buffer.startIndex...index)
    }
    return lines
  }
}
```

```swift
import Foundation

/// mMac's side of the backend protocol: Requests and Replies, Events, and Asks.
public actor BackendClient {
  /// Every `AppEvent` from the backend. It finishes when the backend stops.
  public nonisolated let events: AsyncStream<AppEvent>
  private let eventContinuation: AsyncStream<AppEvent>.Continuation
  private let transport: any Transport
  private let answerAsk: @Sendable (Ask) async -> Int
  private var pending: [Int: CheckedContinuation<JSONValue, Error>] = [:]
  private var nextId = 1
  private var reader: Task<Void, Never>?
  private var stopped = false

  /// `answerAsk` shows the Ask and returns the chosen button's index, or -1.
  public init(transport: any Transport, answerAsk: @escaping @Sendable (Ask) async -> Int) {
    self.transport = transport
    self.answerAsk = answerAsk
    let stream = AsyncStream.makeStream(of: AppEvent.self)
    events = stream.stream
    eventContinuation = stream.continuation
  }

  /// Starts reading the backend's lines. Call it once, before the first request.
  public func run() {
    guard reader == nil else { return }
    reader = Task {
      for await line in self.transport.lines { self.receive(line) }
      self.finish()
    }
  }

  /// Sends a Request and returns the Reply's value.
  public func request(_ method: String, _ args: [JSONValue] = []) async throws -> JSONValue {
    if stopped { throw BackendError.stopped }
    let id = nextId
    nextId += 1
    let line = try JSONEncoder().encode(OutgoingRequest(id: id, method: method, args: args))
    return try await withCheckedThrowingContinuation { continuation in
      pending[id] = continuation
      do {
        try transport.send(line)
      } catch {
        pending.removeValue(forKey: id)
        continuation.resume(throwing: BackendError.stopped)
      }
    }
  }

  /// Sends a Request and decodes the Reply's value as `T`.
  public func call<T: Decodable>(_ method: String, _ args: [JSONValue] = []) async throws -> T {
    try await request(method, args).decode(T.self)
  }

  /// Waits for the backend to exit. Force-stops it after `timeout`.
  public func waitForExit(timeout: Duration) async {
    guard let reader else { return }
    let transport = self.transport
    await withTaskGroup(of: Bool.self) { group in
      group.addTask {
        await reader.value
        return true
      }
      group.addTask {
        try? await Task.sleep(for: timeout)
        return false
      }
      if await group.next() == false { transport.forceStop() }
      group.cancelAll()
    }
  }

  private func receive(_ line: Data) {
    let incoming: Incoming
    do {
      incoming = try Incoming.decode(line)
    } catch {
      eventContinuation.yield(.error(BackendError.unreadable("\(error)").localizedDescription))
      return
    }
    switch incoming {
    case .reply(let id, let result):
      pending.removeValue(forKey: id)?.resume(with: result.mapError { $0 as Error })
    case .event(let event):
      eventContinuation.yield(event)
    case .ask(let ask):
      Task { [answerAsk, transport] in
        let choice = await answerAsk(ask)
        try? transport.send(JSONEncoder().encode(OutgoingAnswer(answer: ask.ask, choice: choice)))
      }
    }
  }

  private func finish() {
    stopped = true
    for continuation in pending.values { continuation.resume(throwing: BackendError.stopped) }
    pending.removeAll()
    eventContinuation.finish()
  }
}
```

```swift
/// Runs `node --import tsx src/headless/server.ts` in a checkout of the fork.
public final class ProcessTransport: Transport, @unchecked Sendable {
  public let lines: AsyncStream<Data>
  private let process = Process()
  private let input = Pipe()
  private let output = Pipe()

  /// Splits stdout into lines. The pipe's handler runs on a background queue.
  private final class Reader: @unchecked Sendable {
    private let lock = NSLock()
    private var splitter = LineSplitter()
    func append(_ data: Data) -> [Data] { lock.withLock { splitter.append(data) } }
  }

  public init(node: URL, repository: URL, environment: [String: String]) throws {
    // A write to a backend that has exited must throw, not end mMac with SIGPIPE.
    signal(SIGPIPE, SIG_IGN)
    let stream = AsyncStream.makeStream(of: Data.self)
    lines = stream.stream
    process.executableURL = node
    process.arguments = ["--import", "tsx", "src/headless/server.ts"]
    process.currentDirectoryURL = repository
    process.environment = environment
    process.standardInput = input
    process.standardOutput = output
    process.standardError = FileHandle.standardError
    let reader = Reader()
    let continuation = stream.continuation
    output.fileHandleForReading.readabilityHandler = { handle in
      let data = handle.availableData
      if data.isEmpty {
        handle.readabilityHandler = nil
        continuation.finish()
        return
      }
      for line in reader.append(data) { continuation.yield(line) }
    }
    try process.run()
  }

  public func send(_ line: Data) throws {
    try input.fileHandleForWriting.write(contentsOf: line + Data([0x0A]))
  }

  public func close() {
    try? input.fileHandleForWriting.close()
  }

  /// Sends SIGTERM, which runs the backend's normal quit path, then SIGKILL after 2 seconds.
  public func forceStop() {
    guard process.isRunning else { return }
    process.terminate()
    let process = self.process
    DispatchQueue.global().asyncAfter(deadline: .now() + 2) {
      if process.isRunning { kill(process.processIdentifier, SIGKILL) }
    }
  }
}
```

```swift
import Foundation

/// Finds Node and starts the backend from a checkout of the fork.
public enum Launcher {
  /// Starts `src/headless/server.ts` in `repository`. Blocks for up to 6 seconds while it reads
  /// the login shell's PATH, so call it off the main actor.
  public static func launch(repository: URL) throws -> ProcessTransport {
    var environment = ProcessInfo.processInfo.environment
    let path = mergePaths([
      environment["PATH"], loginPath(), "/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin",
    ])
    environment["PATH"] = path
    guard let node = findNode(path: path) else {
      throw BackendError.backend("mMac could not find Node. Install Node 22.12 or newer, then restart mMac.")
    }
    return try ProcessTransport(node: node, repository: repository, environment: environment)
  }

  /// The login shell's PATH. An app opened from Finder gets a short PATH without Node or muse.
  static func loginPath() -> String? {
    let shell = ProcessInfo.processInfo.environment["SHELL"] ?? "/bin/zsh"
    let process = Process()
    process.executableURL = URL(fileURLWithPath: shell)
    process.arguments = ["-ilc", #"printf '\0MMAC_PATH\0%s' "$PATH""#]
    let output = Pipe()
    process.standardOutput = output
    process.standardError = FileHandle.nullDevice
    process.standardInput = FileHandle.nullDevice
    do { try process.run() } catch { return nil }
    DispatchQueue.global().asyncAfter(deadline: .now() + 6) {
      if process.isRunning { process.terminate() }
    }
    let data = output.fileHandleForReading.readDataToEndOfFile()
    process.waitUntilExit()
    return parseLoginPath(data)
  }

  /// The text after the `MMAC_PATH` marker. Shell start-up output before it is ignored.
  static func parseLoginPath(_ data: Data) -> String? {
    let marker = Data([0]) + Data("MMAC_PATH".utf8) + Data([0])
    guard let range = data.range(of: marker) else { return nil }
    let value = String(decoding: data[range.upperBound...], as: UTF8.self)
    return value.isEmpty ? nil : value
  }

  /// Joins PATH lists, keeping the first copy of each folder.
  static func mergePaths(_ paths: [String?]) -> String {
    var seen = Set<String>()
    var result: [String] = []
    for folder in paths.compactMap({ $0 }).flatMap({ $0.split(separator: ":").map(String.init) })
    where seen.insert(folder).inserted {
      result.append(folder)
    }
    return result.joined(separator: ":")
  }

  static func findNode(
    path: String, isExecutable: (String) -> Bool = FileManager.default.isExecutableFile(atPath:)
  ) -> URL? {
    for folder in path.split(separator: ":") where isExecutable("\(folder)/node") {
      return URL(fileURLWithPath: "\(folder)/node")
    }
    return nil
  }
}
```

```swift
import MuseBackend

@MainActor @Observable
final class AppModel {
  enum Phase: Equatable {
    case starting
    case running
    case stopped(String)
  }

  var phase: Phase = .starting
  var status: ConnectionStatus?
  var sessions: [Session] = []
  var selected: SessionView?
  var draft = ""
  var errorMessage: String?
  private var client: BackendClient?
  private var quitting = false

  func start() async {
    phase = .starting
    errorMessage = nil
    do {
      let repository = try Self.repository()
      let transport = try await Task.detached { try Launcher.launch(repository: repository) }.value
      let client = BackendClient(transport: transport, answerAsk: { ask in await AppModel.show(ask) })
      self.client = client
      await client.run()
      Task {
        for await event in client.events { self.handle(event) }
        self.backendStopped(client)
      }
      _ = try await client.hello()
      status = try await client.start(dataFolder: Self.dataFolder()).status
      phase = .running
      await refreshSessions()
    } catch {
      phase = .stopped(error.localizedDescription)
    }
  }

    guard let client else { return }
    await run { self.sessions = try await client.sessions().sessions }

    let panel = NSOpenPanel()
    panel.canChooseDirectories = true
    panel.canChooseFiles = false

      let project = try await client.addProject(folder)
      self.selected = try await client.create(project: project)
    }
  }

  func open(_ sessionId: String) async {
    guard let client, sessionId != selected?.session.sessionId else { return }
    await run { self.selected = try await client.open(sessionId) }
  }

  func send() async {
    let text = draft.trimmingCharacters(in: .whitespacesAndNewlines)
    guard let client, let sessionId = selected?.session.sessionId, !text.isEmpty else { return }
    draft = ""
    await run { try await client.send(sessionId: sessionId, text: text) }
  }

  /// Returns true when mMac may quit.
  func quit() async -> Bool {
    guard let client, phase == .running else { return true }
    quitting = true
    do {
      let quit = try await client.shutdown()
      if !quit { quitting = false }
      return quit
    } catch {
      return true
    }
  }

  private func handle(_ event: AppEvent) {
    switch event {
    case .connection(let status): self.status = status
    case .session(let view):
      if view.session.sessionId == selected?.session.sessionId { selected = view }
    case .sessionsChanged: Task { await refreshSessions() }
    case .error(let message): errorMessage = message
    case .other: break
    }
  }

  private func backendStopped(_ stoppedClient: BackendClient) {
    guard client === stoppedClient, !quitting else { return }
    client = nil
    phase = .stopped("The backend process ended. Restart it to continue. Your conversations are saved in Muse Code.")
  }

  private func run(_ work: () async throws -> Void) async {
    do { try await work() } catch { errorMessage = error.localizedDescription }
  }

  /// Shows an Ask as a native alert. The last button cancels and is the default, as in Electron.
  private static func show(_ ask: Ask) -> Int {
    let alert = NSAlert()
    alert.messageText = ask.title
    alert.informativeText = [ask.message, ask.detail].compactMap { $0 }.joined(separator: "\n\n")
    for title in ask.buttons { alert.addButton(withTitle: title) }
    alert.buttons.first?.keyEquivalent = ""
    alert.buttons.last?.keyEquivalent = "\r"
    return alert.runModal().rawValue - NSApplication.ModalResponse.alertFirstButtonReturn.rawValue
  }

  private struct SetupError: LocalizedError {
    let errorDescription: String?
  }

  private static func repository() throws -> URL {
    guard let path = Bundle.main.object(forInfoDictionaryKey: "MMacRepository") as? String else {
      throw SetupError(errorDescription: "This mMac build does not name its checkout. Build it with XcodeGen from mac/project.yml.")
    }
    let repository = URL(fileURLWithPath: path).standardizedFileURL
    guard FileManager.default.fileExists(atPath: repository.appending(path: "src/headless/server.ts").path) else {
      throw SetupError(errorDescription: "mMac could not find src/headless/server.ts in \(repository.path).")
    }
    return repository
  }

  /// `~/Library/Application Support/mMac`. `MMAC_DATA_FOLDER` overrides it for test runs.
  private static func dataFolder() throws -> URL {
    if let folder = ProcessInfo.processInfo.environment["MMAC_DATA_FOLDER"], !folder.isEmpty {
      return URL(fileURLWithPath: folder)
    }
    return try FileManager.default
      .url(for: .applicationSupportDirectory, in: .userDomainMask, appropriateFor: nil, create: true)
      .appending(path: "mMac")
  }
}
```
